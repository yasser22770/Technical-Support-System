import streamlit as st
from datetime import datetime
import random
import sqlite3
import pandas as pd

# 1. إعداد قاعدة البيانات المحلية (SQLite)
conn = sqlite3.connect('enterprise_helpdesk_v8.db', check_same_thread=False)
cursor = conn.cursor()

cursor.execute('''
    CREATE TABLE IF NOT EXISTS tickets (
        ticket_id TEXT PRIMARY KEY,
        customer_name TEXT,
        device_sn TEXT,
        device_type TEXT,
        brand TEXT,
        specs_summary TEXT,
        issues_summary TEXT,
        tech_notes TEXT,
        priority TEXT,
        decision TEXT,
        status TEXT,
        ticket_date TEXT,
        technician_name TEXT,
        cost_est TEXT
    )
''')
conn.commit()

# 2. إعدادات الصفحة واتجاه العرض العربي (RTL)
st.set_page_config(
    page_title="نظام إدارة الدعم الفني وتشخيص الأعطال",
    page_icon="🛠️",
    layout="wide"
)

st.markdown(
    """
    <style>
    .stApp {
        direction: rtl;
        text-align: right;
    }
    </style>
    """,
    unsafe_allow_html=True
)

# 3. القائمة الجانبية للأدوار
st.sidebar.header("🔐 نظام الصلاحيات والأدوار")
user_role = st.sidebar.selectbox(
    "اختر صفتك / دورك في النظام",
    [
        "فني صيانة (Technician)",
        "مدير النظام (IT Manager & Supervisor)",
        "نبذة عن المشروع (CV Overview)"
    ]
)
st.sidebar.markdown("---")

# ==========================================
# الواجهة الأولى: فني الصيانة
# ==========================================
if user_role == "فني صيانة (Technician)":
    st.title("🛠️ لوحة فني الدعم الفني وتشخيص الأعطال")
    st.caption("تسجيل مواصفات الأجهزة، تحديد الأعطال بدقة، وإصدار توصيات فنية واقعية.")
    st.markdown("---")

    st.sidebar.header("🎫 بيانات التذكرة الحالية")
    ticket_id = f"TK-{random.randint(10000, 99999)}"
    st.sidebar.info(f"**رقم التذكرة:** `{ticket_id}`")

    customer_name = st.sidebar.text_input("اسم العميل / القسم", "سارة خالد - قسم الموارد البشرية")
    device_sn = st.sidebar.text_input("الرقم التسلسلي للجهاز (S/N)", "SN-99281044").strip()
    tech_person = st.sidebar.text_input("اسم الفني المسؤول", "ياسر الحازمي")
    priority_level = st.sidebar.selectbox("أولوية التذكرة", ["عادية", "متوسطة", "عاجلة"])
    ticket_date = st.sidebar.date_input("تاريخ الفحص", datetime.now())

    if device_sn:
        cursor.execute("SELECT COUNT(*), MAX(ticket_date) FROM tickets WHERE device_sn = ?", (device_sn,))
        prev_count, last_date = cursor.fetchone()
        if prev_count and prev_count > 0:
            st.warning(f"⚠️ تنبيه: هذا الجهاز (`{device_sn}`) له سجل سابق في النظام ({prev_count} تذكرة سابقة، آخر فحص: {last_date}).")

    col_input1, col_input2 = st.columns([1, 1])

    with col_input1:
        st.subheader("1️⃣ مواصفات وعتاد الجهاز")
        device_type = st.selectbox("نوع الجهاز", ["حاسب محمول (Laptop)", "حاسب مكتبي (Desktop)", "محطة عمل (Workstation)"])
        brand = st.selectbox("الشركة المصنعة", ["Dell", "HP", "Lenovo", "Apple", "Asus", "أخرى"])
        device_age = st.slider("عمر الجهاز (بالسنوات)", 0, 15, 3)

        c1, c2 = st.columns(2)
        with c1:
            ram = st.selectbox("حجم الذاكرة (RAM)", ["4 GB", "8 GB", "16 GB", "32 GB", "64 GB"], index=1)
        with c2:
            storage = st.selectbox("نوع وحدة التخزين", ["HDD تقليدي", "SATA SSD", "NVMe M.2 SSD"])

    with col_input2:
        st.subheader("2️⃣ تحديد الأعطال والمشاكل")
        tab1, tab2, tab3 = st.tabs(["⚙️ العتاد", "💻 النظام", "🌐 الشبكة والحماية"])

        with tab1:
            c_slow = st.checkbox("بطء ملحوظ أو تجمد متكرر")
            c_heat = st.checkbox("ارتفاع حرارة أو صوت مروحة مرتفع")
            c_bsod = st.checkbox("شاشة زرقاء (BSOD) أو إعادة تشغيل")
            c_noise = st.checkbox("أصوات طقطقة ميكانيكية من القرص الصلب")

        with tab2:
            c_boot = st.checkbox("تعذر الإقلاع أو تلف ملفات النظام")
            c_virus = st.checkbox("اشتباه برمجيات خبيثة أو إعلانات مزعجة")
            c_corrupt = st.checkbox("مشكلة في تعريف أو برنامج معين")

        with tab3:
            c_net = st.checkbox("انقطاع Wi-Fi أو ضعف الاتصال")
            c_domain = st.checkbox("مشكلة دخول الشبكة المحلية / النطاق")

    st.subheader("3️⃣ ملاحظات الفني الميدانية")
    tech_notes = st.text_area(
        "أضف أي ملاحظات إضافية لاحظتها أثناء الفحص:",
        placeholder="مثال: تم فحص منافذ USB والشاحن، ويُنصح بترقية التخزين فقط دون تغيير الجهاز..."
    )

    st.markdown("---")

    if st.button("🚀 تنفيذ التشخيص وحفظ التذكرة", type="primary", use_container_width=True):
        selected_issues = []
        if c_slow: selected_issues.append("بطء/تجمد")
        if c_heat: selected_issues.append("حرارة/مروحة")
        if c_bsod: selected_issues.append("شاشة زرقاء")
        if c_noise: selected_issues.append("طقطقة قرص صلب")
        if c_boot: selected_issues.append("تعذر الإقلاع")
        if c_virus: selected_issues.append("برمجيات خبيثة")
        if c_corrupt: selected_issues.append("تعريف/برنامج")
        if c_net: selected_issues.append("شبكة/Wi-Fi")
        if c_domain: selected_issues.append("النطاق المحلي")

        issues_text = "، ".join(selected_issues) if selected_issues else "لا توجد أعطال محددة"
        specs_text = f"{ram} | {storage} | عمر {device_age} سنة"
        notes_clean = tech_notes.strip() if tech_notes.strip() else "لا توجد ملاحظات إضافية"

        # منطق تشخيص واقعي وعادل
        decision = "صيانة برمجية ووقائية"
        cost_estimate = "0 - 100 ر.س"
        time_estimate = "1 - 2 ساعة"

        if c_noise or (device_age >= 8 and c_boot):
            decision = "استبدال الجهاز"
            cost_estimate = "2,500 - 4,800 ر.س"
            time_estimate = "1 - 3 أيام"
        elif storage == "HDD تقليدي" and (c_slow or c_boot):
            decision = "ترقية عتاد (SSD)"
            cost_estimate = "180 - 380 ر.س"
            time_estimate = "2 - 3 ساعات"
        elif "4 GB" in ram and c_slow:
            decision = "ترقية عتاد (RAM)"
            cost_estimate = "120 - 250 ر.س"
            time_estimate = "1 - 2 ساعة"
        elif c_heat:
            decision = "صيانة ميكانيكية وتنظيف حراري"
            cost_estimate = "50 - 150 ر.س"
            time_estimate = "1 - 2 ساعة"

        cursor.execute('''
            INSERT OR REPLACE INTO tickets VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        ''', (
            ticket_id, customer_name, device_sn, device_type, brand,
            specs_text, issues_text, notes_clean, priority_level, decision,
            "مفتوحة", str(ticket_date), tech_person, cost_estimate
        ))
        conn.commit()

        st.subheader(f"📋 تقرير الفحص — التذكرة: {ticket_id}")
        r1, r2, r3, r4 = st.columns(4)
        r1.metric("القرار المقترح", decision)
        r2.metric("الأولوية", priority_level)
        r3.metric("التكلفة التقديرية", cost_estimate)
        r4.metric("الوقت المتوقع", time_estimate)

        st.markdown("---")
        if decision == "استبدال الجهاز":
            st.error("🚨 **التوصية:** يُفضل استبدال الجهاز نظراً لوجود عطل ميكانيكي حرج أو انتهاء عمره التشغيلي.")
        elif "ترقية" in decision:
            st.warning("⚠️ **التوصية:** ترقية القطع المحددة ستحسّن أداء الجهاز بشكل ملحوظ بتكلفة مناسبة.")
        else:
            st.success("✅ **التوصية:** الجهاز لا يحتاج لاستبدال؛ صيانة برمجية أو تنظيف وقائي كافٍ.")

        st.markdown("### ✅ قائمة التحقق والخطوات العملية للفني:")
        recs = []
        if storage == "HDD تقليدي" and (c_slow or c_boot):
            recs.append("نسخ البيانات احتياطياً، تركيب قرص SSD، وتثبيت النظام.")
        if "4 GB" in ram and c_slow:
            recs.append("فحص التوافق ورفع الذاكرة العشوائية إلى 8GB أو 16GB.")
        if c_heat:
            recs.append("تنظيف المراوح ومسارات التهوية وتجديد المعجون الحراري.")
        if c_bsod or c_boot:
            recs.append("تشغيل SFC / DISM وفحص الذاكرة والقرص الصلب.")
        if c_virus:
            recs.append("فحص شامل لمكافحة الفيروسات وإزالة البرامج المشبوهة.")
        if c_corrupt:
            recs.append("إعادة تثبيت التعريف أو البرنامج المتضرر من المصدر الرسمي.")
        if c_net or c_domain:
            recs.append("إعادة ضبط إعدادات الشبكة وفحص إعدادات النطاق.")

        if not recs:
            recs.append("إجراء فحص وقائي دوري وتحديثات النظام الأساسية.")

        for step in recs:
            st.checkbox(step, value=False, key=f"chk_{ticket_id}_{step[:12]}")

        st.info(f"**ملاحظات الفني:** {notes_clean}")

        report_content = (
            f"رقم التذكرة: {ticket_id}\n"
            f"التاريخ: {ticket_date}\n"
            f"الأولوية: {priority_level}\n"
            f"العميل: {customer_name}\n"
            f"الفني: {tech_person}\n"
            f"الجهاز: {device_type} - {brand} ({device_sn})\n"
            f"المواصفات: {specs_text}\n"
            f"الأعطال: {issues_text}\n"
            f"القرار: {decision}\n"
            f"التكلفة التقديرية: {cost_estimate}\n"
            f"الوقت المتوقع: {time_estimate}\n"
            f"ملاحظات الفني: {notes_clean}\n\n"
            f"خطوات التنفيذ:\n" + "\n".join([f"- {x}" for x in recs])
        )

        st.download_button(
            label="📥 تحميل التقرير (.txt)",
            data=report_content,
            file_name=f"Ticket_{ticket_id}.txt",
            mime="text/plain"
        )
        st.success("✅ تم حفظ التذكرة بنجاح في قاعدة البيانات.")

# ==========================================
# الواجهة الثانية: مدير النظام
# ==========================================
elif user_role == "مدير النظام (IT Manager & Supervisor)":
    st.title("📊 لوحة تحكم مدير النظام")
    st.caption("متابعة التذاكر، معاينة التفاصيل، تحديث الحالات والملاحظات، وتصدير السجلات.")
    st.markdown("---")

    df = pd.read_sql_query("SELECT * FROM tickets", conn)

    if df.empty:
        st.info("📭 لا توجد تذاكر مسجلة حالياً.")
        if st.button("✨ تعبئة بيانات تجريبية جاهزة للاستعراض"):
            load_demo_data()
            st.rerun()
    else:
        m1, m2, m3, m4 = st.columns(4)
        m1.metric("إجمالي التذاكر", len(df))
        m2.metric("تذاكر مفتوحة", len(df[df["status"] == "مفتوحة"]))
        m3.metric("تذاكر مكتملة", len(df[df["status"] == "مكتملة"]))
        m4.metric("تذاكر عاجلة", len(df[df["priority"] == "عاجلة"]))

        st.markdown("---")
        ch1, ch2 = st.columns(2)
        with ch1:
            st.markdown("**توزيع التذاكر حسب القرار**")
            st.bar_chart(df["decision"].value_counts())
        with ch2:
            st.markdown("**توزيع التذاكر حسب الحالة**")
            st.bar_chart(df["status"].value_counts())

        repeated_sn = df["device_sn"].value_counts()
        repeated_sn = repeated_sn[repeated_sn > 1]
        if not repeated_sn.empty:
            st.warning("🔁 **أجهزة ذات تذاكر متكررة:** " + "، ".join([f"{sn} ({cnt} مرات)" for sn, cnt in repeated_sn.items()]))

        st.markdown("---")
        f1, f2, f3, f4 = st.columns(4)
        search_term = f1.text_input("🔎 بحث برقم التذكرة أو العميل أو S/N")
        decision_filter = f2.selectbox(
            "تصفية حسب القرار",
            ["الكل"] + sorted(df["decision"].unique().tolist())
        )
        status_filter = f3.selectbox(
            "تصفية حسب الحالة",
            ["الكل"] + sorted(df["status"].unique().tolist())
        )
        tech_filter = f4.selectbox(
            "تصفية حسب الفني",
            ["الكل"] + sorted(df["technician_name"].unique().tolist())
        )

        filtered_df = df.copy()
        if search_term:
            mask = (
                filtered_df["ticket_id"].str.contains(search_term, case=False, na=False) |
                filtered_df["customer_name"].str.contains(search_term, case=False, na=False) |
                filtered_df["device_sn"].str.contains(search_term, case=False, na=False)
            )
            filtered_df = filtered_df[mask]

        if decision_filter != "الكل":
            filtered_df = filtered_df[filtered_df["decision"] == decision_filter]

        if status_filter != "الكل":
            filtered_df = filtered_df[filtered_df["status"] == status_filter]

        if tech_filter != "الكل":
            filtered_df = filtered_df[filtered_df["technician_name"] == tech_filter]

        st.subheader("📋 سجل التذاكر والأجهزة")
        st.dataframe(filtered_df, use_container_width=True)

        st.markdown("### 🔍 معاينة وإدارة تذكرة محددة")
        selected_ticket = st.selectbox("اختر التذكرة", df["ticket_id"].tolist())
        row = df[df["ticket_id"] == selected_ticket].iloc[0]

        with st.expander("عرض البطاقة الكاملة للتذكرة المختارة", expanded=True):
            d1, d2 = st.columns(2)
            with d1:
                st.write(f"**العميل:** {row['customer_name']}")
                st.write(f"**الجهاز:** {row['device_type']} — {row['brand']} (`{row['device_sn']}`)")
                st.write(f"**المواصفات:** {row['specs_summary']}")
                st.write(f"**الأعطال:** {row['issues_summary']}")
            with d2:
                st.write(f"**القرار:** {row['decision']}")
                st.write(f"**الحالة:** {row['status']} | **الأولوية:** {row['priority']}")
                st.write(f"**الفني:** {row['technician_name']} | **التاريخ:** {row['ticket_date']}")
                st.write(f"**التكلفة التقديرية:** {row['cost_est']}")
            st.info(f"**ملاحظات الفني:** {row['tech_notes']}")

            single_report = (
                f"رقم التذكرة: {row['ticket_id']}\n"
                f"التاريخ: {row['ticket_date']}\n"
                f"الحالة: {row['status']} | الأولوية: {row['priority']}\n"
                f"العميل: {row['customer_name']}\n"
                f"الفني: {row['technician_name']}\n"
                f"الجهاز: {row['device_type']} - {row['brand']} ({row['device_sn']})\n"
                f"المواصفات: {row['specs_summary']}\n"
                f"الأعطال: {row['issues_summary']}\n"
                f"القرار: {row['decision']}\n"
                f"التكلفة التقديرية: {row['cost_est']}\n"
                f"ملاحظات الفني: {row['tech_notes']}\n"
            )
            st.download_button(
                label="📥 تحميل تقرير هذه التذكرة (.txt)",
                data=single_report,
                file_name=f"Report_{row['ticket_id']}.txt",
                mime="text/plain",
                key=f"dl_{row['ticket_id']}"
            )

        u1, u2 = st.columns(2)
        new_status = u1.selectbox("الحالة الجديدة", ["مفتوحة", "قيد المعالجة", "مكتملة"])
        new_priority = u2.selectbox("الأولوية الجديدة", ["عادية", "متوسطة", "عاجلة"], index=["عادية", "متوسطة", "عاجلة"].index(row["priority"]))
        updated_notes = st.text_area("تحديث ملاحظات التذكرة", value=row["tech_notes"])

        b1, b2 = st.columns(2)
        if b1.button("💾 حفظ التعديلات"):
            cursor.execute(
                "UPDATE tickets SET status = ?, priority = ?, tech_notes = ? WHERE ticket_id = ?",
                (new_status, new_priority, updated_notes.strip(), selected_ticket)
            )
            conn.commit()
            st.success(f"تم تحديث التذكرة {selected_ticket}.")
            st.rerun()
        if b2.button("🗑️ حذف التذكرة"):
            cursor.execute("DELETE FROM tickets WHERE ticket_id = ?", (selected_ticket,))
            conn.commit()
            st.warning(f"تم حذف التذكرة {selected_ticket}.")
            st.rerun()

        st.markdown("---")
        c_exp, c_demo, c_clr = st.columns(3)
        csv_data = filtered_df.to_csv(index=False).encode("utf-8-sig")
        c_exp.download_button(
            label="📊 تصدير السجل بصيغة CSV",
            data=csv_data,
            file_name="helpdesk_tickets.csv",
            mime="text/csv"
        )

        if c_demo.button("✨ إضافة بيانات تجريبية"):
            load_demo_data()
            st.rerun()

        if c_clr.button("🗑️ مسح جميع التذاكر", type="secondary"):
            cursor.execute("DELETE FROM tickets")
            conn.commit()
            st.rerun()

# ==========================================
# الواجهة الثالثة: نبذة عن المشروع للسيرة الذاتية
# ==========================================
else:
    st.title("📌 نبذة عن المشروع — نظام تشخيص ودعم الحاسب الآلي")
    st.markdown("---")
    st.write(
        "تطبيق ويب تفاعلي لتسجيل مواصفات الأجهزة وتشخيص أعطال العتاد والنظام والشبكات، "
        "مع إصدار توصيات واقعية بين الصيانة الوقائية، ترقية القطع، أو استبدال الجهاز عند الحاجة."
    )
    st.markdown("### أبرز الخصائص التقنية:")
    st.markdown("- **فصل الأدوار:** واجهة مخصصة لفني الصيانة ولوحة متابعة وإحصاءات لمدير النظام.")
    st.markdown("- **قاعدة بيانات SQLite:** حفظ التذاكر وتحديث حالاتها ورصد تكرار الأعطال على نفس الرقم التسلسلي.")
    st.markdown("- **تقارير وتصدير:** تحميل تقرير فني لكل تذكرة بصيغة `.txt` وتصدير السجلات بصيغة `.csv`.")
